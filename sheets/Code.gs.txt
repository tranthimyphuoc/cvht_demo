/** Chạy 1 lần trong editor Apps Script để tạo đủ tab */
function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(SHEET_HEADERS).forEach(function (name) {
    var sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name);
    var headers = SHEET_HEADERS[name];
    sh.clear();
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    sh.setFrozenRows(1);
  });
  // Xóa Sheet1 mặc định nếu còn trống
  var def = ss.getSheetByName('Sheet1');
  if (def && ss.getSheets().length > 1) {
    try { ss.deleteSheet(def); } catch (e) { /* ignore */ }
  }
}

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = body.action;

    // Optional: if (body.apiKey !== 'YOUR_KEY') return json_({ error: 'Unauthorized' });

    if (action === 'setup') {
      setupSheets();
      return json_({ ok: true, sheets: Object.keys(SHEET_HEADERS) });
    }

    if (action === 'pullAll') {
      var out = {};
      Object.keys(SHEET_HEADERS).forEach(function (name) {
        out[name] = sheetToObjects_(ss.getSheetByName(name));
      });
      return json_(out);
    }

    if (action === 'push') {
      var sh = ss.getSheetByName(body.sheet);
      if (!sh) return json_({ error: 'Sheet not found: ' + body.sheet });
      var headers = body.headers && body.headers.length
        ? body.headers
        : (SHEET_HEADERS[body.sheet] || Object.keys((body.rows && body.rows[0]) || { id: '' }));
      writeObjects_(sh, body.rows || [], headers);
      return json_({ ok: true, count: (body.rows || []).length });
    }

    if (action === 'append') {
      var shA = ss.getSheetByName(body.sheet);
      if (!shA) return json_({ error: 'Sheet not found' });
      appendObject_(shA, body.row, SHEET_HEADERS[body.sheet]);
      return json_({ ok: true });
    }

    return json_({ error: 'Unknown action' });
  } catch (err) {
    return json_({ error: String(err) });
  }
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function sheetToObjects_(sh) {
  if (!sh) return [];
  var values = sh.getDataRange().getValues();
  if (values.length < 2) return [];
  var headers = values[0].map(String);
  return values.slice(1).filter(function (r) {
    return r.some(function (c) { return c !== ''; });
  }).map(function (row) {
    var o = {};
    headers.forEach(function (h, i) { o[h] = row[i]; });
    return o;
  });
}

function writeObjects_(sh, rows, headers) {
  sh.clearContents();
  if (!headers || !headers.length) {
    if (!rows.length) return;
    headers = Object.keys(rows[0]);
  }
  var data = [headers];
  rows.forEach(function (r) {
    data.push(headers.map(function (h) {
      var v = r[h];
      return v == null ? '' : v;
    }));
  });
  sh.getRange(1, 1, data.length, headers.length).setValues(data);
  sh.setFrozenRows(1);
}

function appendObject_(sh, row, fallbackHeaders) {
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(String);
  if (!headers.length || !headers[0]) {
    headers = fallbackHeaders || Object.keys(row || {});
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
  sh.appendRow(headers.map(function (h) {
    return row[h] == null ? '' : row[h];
  }));
}
