/**
 * Tạo HTTP response JSON chuẩn.
 * @param {object} obj - Dữ liệu trả về
 */
function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Response thành công.
 * @param {*} data - Dữ liệu
 * @param {string} [message='Success']
 * @param {number} [statusCode=200]
 */
function successResponse(data, message, statusCode) {
  return jsonResponse_({
    success: true,
    message: message || 'Success',
    data:    data    || null,
    status:  statusCode || 200,
  });
}

/**
 * Response lỗi.
 * @param {string} message - Mô tả lỗi
 * @param {number} [statusCode=400]
 * @param {*} [details] - Chi tiết lỗi (validation errors, v.v.)
 */
function errorResponse(message, statusCode, details) {
  const res = {
    success: false,
    message: message || 'Error',
    status:  statusCode || 400,
  };
  if (details) res.details = details;
  return jsonResponse_(res);
}

// Shortcuts cho các HTTP status phổ biến
const error400 = (msg, details) => errorResponse(msg || 'Bad Request',   400, details);
const error401 = (msg)          => errorResponse(msg || 'Unauthorized',  401);
const error403 = (msg)          => errorResponse(msg || 'Forbidden',     403);
const error404 = (msg)          => errorResponse(msg || 'Not Found',     404);
const error500 = (msg)          => errorResponse(msg || 'Internal Server Error', 500);

/**
 * Lấy Spreadsheet active.
 */
function getSpreadsheet_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

/**
 * Lấy sheet theo tên; tạo mới nếu chưa có.
 */
function getOrCreateSheet_(name) {
  const ss = getSpreadsheet_();
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    const headers = SHEET_HEADERS[name];
    if (headers) {
      sh.getRange(1, 1, 1, headers.length).setValues([headers]);
      sh.setFrozenRows(1);
    }
  }
  return sh;
}

/**
 * Đọc toàn bộ sheet, trả về mảng object {header: value}.
 * @param {string} sheetName
 * @returns {object[]}
 */
function sheetToObjects_(sheetName) {
  const sh = getSpreadsheet_().getSheetByName(sheetName);
  if (!sh) return [];
  const values = sh.getDataRange().getValues();
  if (values.length < 2) return [];
  const headers = values[0].map(String);
  return values.slice(1)
    .filter(row => row.some(c => c !== ''))
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => { obj[h] = row[i]; });
      return obj;
    });
}

/**
 * Ghi đè toàn bộ sheet (xóa cũ, ghi mới).
 * @param {string} sheetName
 * @param {object[]} rows
 * @param {string[]} [headers] - Nếu không truyền, dùng SHEET_HEADERS
 */
function slimAttachmentsJson_(raw) {
  var list = [];
  try {
    if (Array.isArray(raw)) list = raw;
    else if (typeof raw === 'string' && raw.trim()) list = JSON.parse(raw);
    else if (raw && typeof raw === 'object') list = [raw];
  } catch (e) {
    return '[]';
  }
  if (!Array.isArray(list)) return '[]';
  var out = list.map(function(a) {
    a = a || {};
    return {
      id: a.id || '',
      name: a.name || '',
      type: a.type || '',
      size: a.size || '',
      kind: a.kind || '',
      url: a.url || a.viewUrl || '',
      driveId: a.driveId || '',
      downloadUrl: a.downloadUrl || '',
    };
  });
  return JSON.stringify(out);
}

function slimReportRow_(row) {
  if (!row) return row;
  var out = Object.assign({}, row);
  if (out.attachmentsJson != null && out.attachmentsJson !== '') {
    out.attachmentsJson = slimAttachmentsJson_(out.attachmentsJson);
  } else if (out.attachments) {
    out.attachmentsJson = slimAttachmentsJson_(out.attachments);
  }
  return out;
}

function sheetCell_(v) {
  if (v == null) return '';
  if (typeof v === 'string' && v.length > 49900) {
    if (v.indexOf('data:') !== -1) return slimAttachmentsJson_(v);
    return v.substring(0, 49900);
  }
  return v;
}

function writeObjects_(sheetName, rows, headers) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const sh = getOrCreateSheet_(sheetName);
    sh.clearContents();
    const hdrs = headers || SHEET_HEADERS[sheetName] || (rows.length ? Object.keys(rows[0]) : []);
    if (!hdrs.length) return;
    const data = [hdrs, ...rows.map(r => hdrs.map(h => sheetCell_(r[h] == null ? '' : r[h])))];
    sh.getRange(1, 1, data.length, hdrs.length).setValues(data);
    sh.setFrozenRows(1);
  } finally {
    lock.releaseLock();
  }
}

/**
 * Append 1 row vào sheet.
 * @param {string} sheetName
 * @param {object} row
 */
function appendObject_(sheetName, row) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const sh = getOrCreateSheet_(sheetName);
    const headers = SHEET_HEADERS[sheetName] || Object.keys(row);
    sh.appendRow(headers.map(h => sheetCell_(row[h] == null ? '' : row[h])));
  } finally {
    lock.releaseLock();
  }
}

/**
 * Đảm bảo dòng header có đủ cột needed.
 */
function ensureSheetHeaders_(sh, needed) {
  needed = needed || [];
  var lastCol = sh.getLastColumn();
  var lastRow = sh.getLastRow();
  var existing = [];
  if (lastRow >= 1 && lastCol >= 1) {
    existing = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(String);
    while (existing.length && existing[existing.length - 1] === '') existing.pop();
  }
  if (!existing.length) {
    if (needed.length) {
      sh.getRange(1, 1, 1, needed.length).setValues([needed]);
      sh.setFrozenRows(1);
    }
    return needed.slice();
  }
  var added = false;
  needed.forEach(function(h) {
    if (h && existing.indexOf(h) === -1) {
      existing.push(h);
      added = true;
    }
  });
  if (added) {
    sh.getRange(1, 1, 1, existing.length).setValues([existing]);
  }
  return existing;
}

/**
 * Upsert: tìm theo field, update nếu tồn tại, append nếu chưa có.
 */
function upsertObject_(sheetName, row, keyField) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const key = keyField || 'id';
    const sh  = getOrCreateSheet_(sheetName);
    const needed = SHEET_HEADERS[sheetName] || Object.keys(row);
    const headers = ensureSheetHeaders_(sh, needed);
    const values  = sh.getDataRange().getValues();

    if (values.length < 1) {
      sh.appendRow(headers.map(h => sheetCell_(row[h] == null ? '' : row[h])));
      return;
    }
    const hdrs = values[0].map(String);
    const keyIdx = hdrs.indexOf(key);
    if (keyIdx === -1) {
      sh.appendRow(headers.map(h => sheetCell_(row[h] == null ? '' : row[h])));
      return;
    }

    for (var i = 1; i < values.length; i++) {
      if (String(values[i][keyIdx]) === String(row[key])) {
        var out = hdrs.map(function(h) {
          var val = row[h] !== undefined ? row[h] : values[i][hdrs.indexOf(h)];
          return sheetCell_(val);
        });
        sh.getRange(i + 1, 1, 1, hdrs.length).setValues([out]);
        return;
      }
    }
    sh.appendRow(headers.map(h => sheetCell_(row[h] == null ? '' : row[h])));
  } finally {
    lock.releaseLock();
  }
}

/**
 * Đặt active cho 1 user theo id — chỉ sửa 1 ô, không rewrite cả sheet.
 * @returns {boolean} true nếu tìm thấy và đã ghi
 */
function setUserActive_(userId, active) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const sh = getSpreadsheet_().getSheetByName('Users');
    if (!sh) return false;
    const values = sh.getDataRange().getValues();
    if (values.length < 2) return false;
    const hdrs = values[0].map(String);
    const idIdx = hdrs.indexOf('id');
    const activeIdx = hdrs.indexOf('active');
    if (idIdx < 0 || activeIdx < 0) return false;
    const want = String(userId || '').trim();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][idIdx]).trim() === want) {
        sh.getRange(i + 1, activeIdx + 1).setValue(active ? 'TRUE' : 'FALSE');
        var updIdx = hdrs.indexOf('updatedAt');
        if (updIdx >= 0) sh.getRange(i + 1, updIdx + 1).setValue(new Date().toISOString());
        return true;
      }
    }
    return false;
  } finally {
    lock.releaseLock();
  }
}

/**
 * Tìm 1 object theo field=value.
 * @param {string} sheetName
 * @param {string} field
 * @param {*} value
 * @returns {object|null}
 */
function findOne_(sheetName, field, value) {
  return sheetToObjects_(sheetName).find(r => String(r[field]) === String(value)) || null;
}

/**
 * Xóa row theo field=value.
 * @param {string} sheetName
 * @param {string} field
 * @param {*} value
 */
function deleteRow_(sheetName, field, value) {
  const sh = getSpreadsheet_().getSheetByName(sheetName);
  if (!sh) return;
  const values = sh.getDataRange().getValues();
  if (values.length < 2) return;
  const idx = values[0].map(String).indexOf(field);
  if (idx === -1) return;
  // Duyệt từ dưới lên để tránh lệch chỉ số khi xóa
  for (let i = values.length - 1; i >= 1; i--) {
    if (String(values[i][idx]) === String(value)) {
      sh.deleteRow(i + 1);
    }
  }
}

/**
 * Tạo ID duy nhất dạng '<prefix>_<timestamp><random>'.
 * @param {string} [prefix='id']
 */
function generateId_(prefix) {
  const ts  = Date.now().toString(36);
  const rnd = Math.random().toString(36).slice(2, 7);
  return (prefix || 'id') + '_' + ts + rnd;
}

/**
 * Encode chuỗi sang base64url (không có padding, URL-safe).
 * @param {string} str
 */
function base64UrlEncode_(str) {
  return Utilities.base64Encode(str, Utilities.Charset.UTF_8)
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
}

/**
 * Decode base64url về chuỗi UTF-8.
 * @param {string} b64url
 */
function base64UrlDecode_(b64url) {
  let s = b64url.replace(/-/g, '+').replace(/_/g, '/');
  while (s.length % 4) s += '=';
  return Utilities.newBlob(Utilities.base64Decode(s)).getDataAsString();
}

/**
 * Ghi log request vào sheet Logs.
 * @param {object} entry - { ip, method, endpoint, userId, role, status, durationMs, userAgent }
 */
function writeLog_(entry) {
  try {
    appendObject_('Logs', {
      id:         generateId_('log'),
      time:       new Date().toISOString(),
      ip:         entry.ip         || '',
      method:     entry.method     || '',
      endpoint:   entry.endpoint   || '',
      userId:     entry.userId     || '',
      role:       entry.role       || '',
      status:     entry.status     || '',
      durationMs: entry.durationMs || 0,
      userAgent:  entry.userAgent  || '',
    });
  } catch (e) {
    // Không để lỗi log làm crash cả request
    console.error('Log error:', e);
  }
}

/**
 * Kiểm tra rate limit theo IP, dùng Cache Service (1 phút).
 * Trả về true nếu vượt giới hạn.
 * @param {string} ip
 */
function isRateLimited_(ip) {
  if (!ip) return false;
  try {
    const cache   = CacheService.getScriptCache();
    const key     = 'rl_' + ip.replace(/[^a-z0-9]/gi, '_');
    const current = parseInt(cache.get(key) || '0', 10);
    if (current >= CONFIG.MAX_REQUESTS_PER_MINUTE) return true;
    cache.put(key, String(current + 1), 60); // TTL 60 giây
    return false;
  } catch (e) {
    return false; // Nếu Cache lỗi, cho qua
  }
}

/**
 * Kiểm tra origin có được phép không.
 * @param {string} origin
 */
function isOriginAllowed_(origin) {
  if (!origin) return true; // Không có origin header (server-to-server)
  return CONFIG.ALLOWED_ORIGINS.some(o => origin.startsWith(o));
}

/**
 * Lấy IP từ request headers (Apps Script không expose IP trực tiếp).
 * Dùng X-Forwarded-For nếu có.
 * @param {object} headers
 */
function getClientIp_(headers) {
  if (!headers) return 'unknown';
  return (headers['X-Forwarded-For'] || headers['x-forwarded-for'] || 'unknown').split(',')[0].trim();
}

/**
 * Làm sạch chuỗi: strip HTML tags, trim.
 * Chống XSS cơ bản.
 * @param {string} str
 */
function sanitize_(str) {
  if (typeof str !== 'string') return str;
  return str
    .replace(/<[^>]*>/g, '')        // strip HTML tags
    .replace(/[<>"'&]/g, c => ({    // escape special chars
      '<': '&lt;', '>': '&gt;',
      '"': '&quot;', "'": '&#39;',
      '&': '&amp;',
    })[c] || c)
    .trim();
}

/**
 * Sanitize đệ quy một object.
 * @param {object|string} input
 */
function getAttachmentsFolder_() {
  var name = 'CVHT-Hub-Attachments';
  try {
    var ss = getSpreadsheet_();
    var file = DriveApp.getFileById(ss.getId());
    var parents = file.getParents();
    var parent = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
    var it = parent.getFoldersByName(name);
    if (it.hasNext()) return it.next();
    return parent.createFolder(name);
  } catch (err) {
    Logger.log('getAttachmentsFolder_ fallback: ' + err);
    var rootIt = DriveApp.getRootFolder().getFoldersByName(name);
    if (rootIt.hasNext()) return rootIt.next();
    return DriveApp.getRootFolder().createFolder(name);
  }
}

/**
 * Lưu 1 file (base64) lên Google Drive — không nhét vào ô Sheet (giới hạn 50k ký tự).
 */
function uploadAttachmentFile_(name, mimeType, base64Data) {
  if (!base64Data) throw new Error('Thiếu dữ liệu file');
  var folder = getAttachmentsFolder_();
  var decoded = Utilities.base64Decode(String(base64Data));
  var blob = Utilities.newBlob(
    decoded,
    mimeType || 'application/octet-stream',
    name || ('file-' + Date.now())
  );
  var file = folder.createFile(blob);
  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (err) {
    Logger.log('setSharing: ' + err);
  }
  var id = file.getId();
  return {
    fileId: id,
    url: 'https://drive.google.com/file/d/' + id + '/view',
    downloadUrl: 'https://drive.google.com/uc?export=download&id=' + id,
    viewUrl: 'https://drive.google.com/uc?export=view&id=' + id,
  };
}

function sanitizeInput_(input) {
  if (typeof input === 'string') return sanitize_(input);
  if (Array.isArray(input))     return input.map(sanitizeInput_);
  if (input && typeof input === 'object') {
    const out = {};
    Object.keys(input).forEach(k => { out[k] = sanitizeInput_(input[k]); });
    return out;
  }
  return input;
}
